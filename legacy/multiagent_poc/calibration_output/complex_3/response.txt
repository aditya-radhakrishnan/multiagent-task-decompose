```python
from flask import Flask
from flask_sqlalchemy import SQLAlchemy
from flask_migrate import Migrate
from flask_jwt_extended import JWTManager
from flask_cors import CORS
from celery import Celery
import redis

db = SQLAlchemy()
migrate = Migrate()
jwt = JWTManager()
celery = Celery(__name__)
redis_client = redis.Redis()

def create_app(config_class='app.config.Config'):
    app = Flask(__name__)
    app.config.from_object(config_class)
    
    # Initialize extensions
    db.init_app(app)
    migrate.init_app(app, db)
    jwt.init_app(app)
    CORS(app)
    
    # Configure Celery
    celery.conf.update(
        broker_url=app.config['CELERY_BROKER_URL'],
        result_backend=app.config['CELERY_RESULT_BACKEND']
    )
    
    # Register blueprints
    from app.api import bp as api_bp
    app.register_blueprint(api_bp, url_prefix='/api')
    
    return app

def make_celery(app):
    celery.conf.update(app.config)
    
    class ContextTask(celery.Task):
        def __call__(self, *args, **kwargs):
            with app.app_context():
                return self.run(*args, **kwargs)
    
    celery.Task = ContextTask
    return celery
```

### backend/app/services/data_service.py
```python
import pandas as pd
import requests
import psycopg2
from sqlalchemy import create_engine
from app.models import DataSource
import io
import json

class DataService:
    @staticmethod
    def get_data(data_source_id: int, query: dict = None):
        """Get data from a data source"""
        data_source = DataSource.query.get(data_source_id)
        if not data_source:
            raise ValueError("Data source not found")
        
        config = data_source.get_config()
        
        if data_source.type == 'csv':
            return DataService._get_csv_data(config, query)
        elif data_source.type == 'database':
            return DataService._get_database_data(config, query)
        elif data_source.type == 'api':
            return DataService._get_api_data(config, query)
        else:
            raise ValueError(f"Unsupported data source type: {data_source.type}")
    
    @staticmethod
    def _get_csv_data(config: dict, query: dict = None):
        """Get data from CSV file"""
        if 'file_path' in config:
            df = pd.read_csv(config['file_path'])
        elif 'url' in config:
            df = pd.read_csv(config['url'])
        else:
            raise ValueError("CSV data source requires file_path or url")
        
        # Apply filters if provided
        if query and 'filters' in query:
            for filter_item in query['filters']:
                column = filter_item['column']
                operator = filter_item['operator']
                value = filter_item['value']
                
                if operator == 'eq':
                    df = df[df[column] == value]
                elif operator == 'gt':
                    df = df[df[column] > value]
                elif operator == 'lt':
                    df = df[df[column] < value]
        
        # Apply limit
        if query and 'limit' in query:
            df = df.head(query['limit'])
        
        return df.to_dict('records')
    
    @staticmethod
    def _get_database_data(config: dict, query: dict = None):
        """Get data from database"""
        engine = create_engine(config['connection_string'])
        
        sql_query = config.get('query', 'SELECT * FROM ' + config.get('table', ''))
        
        if query and 'sql' in query:
            sql_query = query['sql']
        
        df = pd.read_sql(sql_query, engine)
        return df.to_dict('records')
    
    @staticmethod
    def _get_api_data(config: dict, query: dict = None):
        """Get data from API"""
        url = config['url']
        headers = config.get('headers', {})
        params = config.get('params', {})
        
        if query and 'params' in query:
            params.update(query['params'])
        
        response = requests.get(url, headers=headers, params=params)
        response.raise_for_status()
        
        data = response.json()
        
        # Handle different response structures
        if isinstance(data, list):
            return data
        elif 'data' in data:
            return data['data']
        else:
            return [data]

class MetricService:
    @staticmethod
    def calculate_metric(metric_id: int, data_source_id: int = None):
        """Calculate a custom metric"""
        from app.models import Metric
        
        metric = Metric.query.get(metric_id)
        if not metric:
            raise ValueError("Metric not found")
        
        source_id = data_source_id or metric.data_source_id
        data = DataService.get_data(source_id)
        
        df = pd.DataFrame(data)
        
        # Simple formula evaluation (in production, use a more secure approach)
        # This is a simplified version - you'd want to implement proper expression parsing
        formula = metric.formula
        
        # Replace common aggregations
        if 'SUM(' in formula:
            column = formula.split('SUM(')[1].split(')')[0]
            return float(df[column].sum())
        elif 'AVG(' in formula:
            column = formula.split('AVG(')[1].split(')')[0]
            return float(df[column].mean())
        elif 'COUNT(' in formula:
            return len(df)
        elif 'MAX(' in formula:
            column = formula.split('MAX(')[1].split(')')[0]
            return float(df[column].max())
        elif 'MIN(' in formula:
            column = formula.split('MIN(')[1].split(')')[0]
            return float(df[column].min())
        
        # For complex formulas, you might want to use eval() with restrictions
        # or implement a proper expression parser
        return 0
```

```python
import pandas as pd
import io
from flask import make_response
from app.services.data_service import DataService

class ExportService:
    @staticmethod
    def export_data(data_source_id: int, format: str = 'csv', query: dict = None):
        """Export data in various formats"""
        data = DataService.get_data(data_source_id, query)
        df = pd.DataFrame(data)
        
        if format == 'csv':
            return ExportService._export_csv(df)
        elif format == 'excel':
            return ExportService._export_excel(df)
        elif format == 'json':
            return ExportService._export_json(df)
        else:
            raise ValueError(f"Unsupported export format: {format}")
    
    @staticmethod
    def _export_csv(df):
        output = io.StringIO()
        df.to_csv(output, index=False)
        
        response = make_response(output.getvalue())
        response.headers["Content-Disposition"] = "attachment; filename=export.csv"
        response.headers["Content-type"] = "text/csv"
        return response
    
    @staticmethod
    def _export_excel(df):
        output = io.BytesIO()
        with pd.ExcelWriter(output, engine='xlsxwriter') as writer:
            df.to_excel(writer, sheet_name='Data', index=False)
        
        response = make_response(output.getvalue())
        response.headers["Content-Disposition"] = "attachment; filename=export.xlsx"
        response.headers["Content-type"] = "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
        return response
    
    @staticmethod
    def _export_json(df):
        json_data = df.to_json(orient='records', indent=2)
        
        response = make_response(json_data)
        response.headers["Content-Disposition"] = "attachment; filename=export.json"
        response.headers["Content-type"] = "application/json"
        return response
```

```python
from flask import request, jsonify
from flask_jwt_extended import create_access_token, jwt_required, get_jwt_identity
from app.api import bp
from app.models import User
from app import db

@bp.route('/auth/register', methods=['POST'])
def register():
    data = request.get_json()
    
    if User.query.filter_by(username=data['username']).first():
        return jsonify({'message': 'Username already exists'}), 400
    
    if User.query.filter_by(email=data['email']).first():
        return jsonify({'message': 'Email already exists'}), 400
    
    user = User(
        username=data['username'],
        email=data['email']
    )
    user.set_password(data['password'])
    
    db.session.add(user)
    db.session.commit()
    
    access_token = create_access_token(identity=user.id)
    
    return jsonify({
        'access_token': access_token,
        'user': user.to_dict()
    })

@bp.route('/auth/login', methods=['POST'])
def login():
    data = request.get_json()
    user = User.query.filter_by(username=data['username']).first()
    
    if user and user.check_password(data['password']):
        access_token = create_access_token(identity=user.id)
        return jsonify({
            'access_token': access_token,
            'user': user.to_dict()
        })
    
    return jsonify({'message': 'Invalid credentials'}), 401

@bp.route('/auth/me', methods=['GET'])
@jwt_required()
def get_current_user():
    user_id = get_jwt_identity()
    user = User.query.get(user_id)
    return jsonify(user.to_dict())
```

```python
    widget.width = data.get('size', {}).get('width', widget.width)
    widget.height = data.get('size', {}).get('height', widget.height)
    
    if 'config' in data:
        widget.set_config(data['config'])
    
    db.session.commit()
    return jsonify(widget.to_dict())

@bp.route('/dashboards/<int:dashboard_id>/widgets/<int:widget_id>', methods=['DELETE'])
@jwt_required()
def delete_widget(dashboard_id, widget_id):
    user_id = get_jwt_identity()
    dashboard = Dashboard.query.filter_by(id=dashboard_id, owner_id=user_id).first()
    
    if not dashboard:
        return jsonify({'message': 'Dashboard not found'}), 404
    
    widget = Widget.query.filter_by(id=widget_id, dashboard_id=dashboard_id).first()
    
    if not widget:
        return jsonify({'message': 'Widget not found'}), 404
    
    db.session.delete(widget)
    db.session.commit()
    
    return jsonify({'message': 'Widget deleted'})
```

```typescript
import React, { useState, useEffect } from 'react';
import {
  Paper,
  Typography,
  Box,
  IconButton,
  Menu,
  MenuItem,
  CircularProgress,
} from '@mui/material';
import { MoreVert as MoreVertIcon, TrendingUp, TrendingDown } from '@mui/icons-material';

import { dataSourceAPI, metricsAPI } from '../../services/api';
import { Widget } from '../../types';

interface MetricWidgetProps {
  widget: Widget;
  onDelete: () => void;
}

const MetricWidget: React.FC<MetricWidgetProps> = ({ widget, onDelete }) => {
  const [value, setValue] = useState<number | null>(null);
  const [loading, setLoading] = useState(true);
  const [anchorEl, setAnchorEl] = useState<null | HTMLElement>(null);

  useEffect(() => {
    loadMetricValue();
  }, [widget.data_source_id, widget.config]);

  const loadMetricValue = async () => {
    if (!widget.data_source_id) return;

    try {
      setLoading(true);
      
      // If metric_id is configured, use the metric calculation
      if (widget.config.metric_id) {
        const response = await metricsAPI.calculate(
          widget.config.metric_id,
          widget.data_source_id
        );
        setValue(response.data.value);
      } else {
        // Otherwise, perform simple aggregation on data
        const response = await dataSourceAPI.getData(widget.data_source_id);
        const data = response.data;
        
        if (data.length > 0 && widget.config.field) {
          const field = widget.config.field;
          const aggregation = widget.config.aggregation || 'sum';
          
          let calculatedValue = 0;
          
          switch (aggregation) {
            case 'sum':
              calculatedValue = data.reduce((acc: number, item: any) => acc + (parseFloat(item[field]) || 0), 0);
              break;
            case 'avg':
              calculatedValue = data.reduce((acc: number, item: any) => acc + (parseFloat(item[field]) || 0), 0) / data.length;
              break;
            case 'count':
              calculatedValue = data.length;
              break;
            case 'max':
              calculatedValue = Math.max(...data.map((item: any) => parseFloat(item[field]) || 0));
              break;
            case 'min':
              calculatedValue = Math.min(...data.map((item: any) => parseFloat(item[field]) || 0));
              break;
            default:
              calculatedValue = 0;
          }
          
          setValue(calculatedValue);
        }
      }
    } catch (error) {
      console.error('Failed to load metric value:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleMenuClick = (event: React.MouseEvent<HTMLElement>) => {
    setAnchorEl(event.currentTarget);
  };

  const handleMenuClose = () => {
    setAnchorEl(null);
  };

  const formatValue = (val: number) => {
    if (val >= 1000000) {
      return `${(val / 1000000).toFixed(1)}M`;
    } else if (val >= 1000) {
      return `${(val / 1000).toFixed(1)}K`;
    }
    return val.toLocaleString();
  };

  const getTrendIcon = () => {
    const trend = widget.config.trend;
    if (trend === 'up') {
      return <TrendingUp sx={{ color: 'success.main' }} />;
    } else if (trend === 'down') {
      return <TrendingDown sx={{ color: 'error.main' }} />;
    }
    return null;
  };

  return (
    <Paper sx={{ p: 3, height: '100%', position: 'relative' }}>
      <Box sx={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start' }}>
        <Box sx={{ flex: 1 }}>
          <Typography variant="subtitle2" color="textSecondary" gutterBottom>
            {widget.name}
          </Typography>
          
          {loading ? (
            <CircularProgress size={24} />
          ) : (
            <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
              <Typography variant="h3" component="div" sx={{ fontWeight: 'bold' }}>
                {value !== null ? formatValue(value) : '-'}
              </Typography>
              {getTrendIcon()}
            </Box>
          )}
          
          {widget.config.subtitle && (
            <Typography variant="body2" color="textSecondary" sx={{ mt: 1 }}>
              {widget.config.subtitle}
            </Typography>
          )}
        </Box>
        
        <IconButton size="small" onClick={handleMenuClick}>
          <MoreVertIcon />
        </IconButton>
      </Box>

      <Menu anchorEl={anchorEl} open={Boolean(anchorEl)} onClose={handleMenuClose}>
        <MenuItem onClick={() => { /* Open config dialog */; handleMenuClose(); }}>
          Configure
        </MenuItem>
        <MenuItem onClick={() => { onDelete(); handleMenuClose(); }}>
          Delete
        </MenuItem>
      </Menu>
    </Paper>
  );
};

export default MetricWidget;
```
