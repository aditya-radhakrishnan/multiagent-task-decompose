// userPreferencesApi.ts

/**
 * User Preferences Response Interface
 */
export interface UserPreferences {
  theme: string;
  language: string;
  notifications: {
    email: boolean;
    push: boolean;
    sms: boolean;
  };
  privacy: {
    profileVisibility: string;
    showEmail: boolean;
    showOnlineStatus: boolean;
  };
  display: {
    timezone: string;
    dateFormat: string;
    timeFormat: string;
  };
}

export interface UserPreferencesResponse {
  userId: string;
  preferences: UserPreferences;
  lastUpdated: string;
}

/**
 * API Error Response Interface
 */
export interface ApiError {
  error: {
    code: string;
    message: string;
  };
}

/**
 * Custom Error Class for API Errors
 */
export class UserPreferencesApiError extends Error {
  code: string;
  statusCode: number;

  constructor(message: string, code: string, statusCode: number) {
    super(message);
    this.name = 'UserPreferencesApiError';
    this.code = code;
    this.statusCode = statusCode;
  }
}

/**
 * Configuration Interface
 */
export interface ApiConfig {
  baseUrl?: string;
  timeout?: number;
}

/**
 * Fetches user preferences from the API
 * 
 * @param token - JWT authentication token
 * @param config - Optional API configuration
 * @returns Promise resolving to UserPreferencesResponse
 * @throws UserPreferencesApiError on failure
 */
export async function getUserPreferences(
  token: string,
  config: ApiConfig = {}
): Promise<UserPreferencesResponse> {
  const { baseUrl = '', timeout = 10000 } = config;
  const url = `${baseUrl}/api/user/preferences`;

  // Validate token
  if (!token || token.trim() === '') {
    throw new UserPreferencesApiError(
      'Authentication token is required',
      'MISSING_TOKEN',
      401
    );
  }

  // Create abort controller for timeout
  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), timeout);

  try {
    const response = await fetch(url, {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json',
      },
      signal: controller.signal,
    });

    clearTimeout(timeoutId);

    // Handle non-2xx responses
    if (!response.ok) {
      const errorData: ApiError = await response.json().catch(() => ({
        error: {
          code: 'UNKNOWN_ERROR',
          message: 'An unexpected error occurred',
        },
      }));

      throw new UserPreferencesApiError(
        errorData.error.message,
        errorData.error.code,
        response.status
      );
    }

    // Parse and return successful response
    const data: UserPreferencesResponse = await response.json();
    return data;

  } catch (error) {
    clearTimeout(timeoutId);

    // Handle UserPreferencesApiError (already thrown above)
    if (error instanceof UserPreferencesApiError) {
      throw error;
    }

    // Handle abort/timeout errors
    if (error instanceof Error && error.name === 'AbortError') {
      throw new UserPreferencesApiError(
        'Request timeout',
        'TIMEOUT',
        408
      );
    }

    // Handle network errors
    if (error instanceof TypeError) {
      throw new UserPreferencesApiError(
        'Network error: Unable to reach server',
        'NETWORK_ERROR',
        0
      );
    }

    // Handle unknown errors
    throw new UserPreferencesApiError(
      error instanceof Error ? error.message : 'An unexpected error occurred',
      'UNKNOWN_ERROR',
      500
    );
  }
}

function UserPreferencesComponent({ authToken }) {
  const [preferences, setPreferences] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    async function fetchPreferences() {
      try {
        const data = await getUserPreferences(authToken);
        setPreferences(data.preferences);
        setError(null);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    }

    fetchPreferences();
  }, [authToken]);

  if (loading) return <div>Loading preferences...</div>;
  if (error) return <div>Error: {error}</div>;
  
  return (
    <div>
      <h2>User Preferences</h2>
      <p>Theme: {preferences.theme}</p>
