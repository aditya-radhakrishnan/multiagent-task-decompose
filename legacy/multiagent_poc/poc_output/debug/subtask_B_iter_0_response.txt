```python
# app.py or routes/user_preferences.py

from flask import Flask, jsonify, request
from functools import wraps
from datetime import datetime
import jwt

app = Flask(__name__)

# Configuration (should be in config file in production)
app.config['SECRET_KEY'] = 'your-secret-key-here'  # Change in production

# Mock database - replace with actual database in production
MOCK_USER_PREFERENCES = {
    "user_12345": {
        "theme": "dark",
        "language": "en",
        "notifications": {
            "email": True,
            "push": True,
            "sms": False
        },
        "privacy": {
            "profileVisibility": "public",
            "showEmail": False,
            "showOnlineStatus": True
        },
        "display": {
            "timezone": "America/New_York",
            "dateFormat": "MM/DD/YYYY",
            "timeFormat": "12h"
        }
    }
}

MOCK_USER_UPDATED_TIMES = {
    "user_12345": "2024-01-15T10:30:00Z"
}

# Default preferences for new users
DEFAULT_PREFERENCES = {
    "theme": "light",
    "language": "en",
    "notifications": {
        "email": True,
        "push": False,
        "sms": False
    },
    "privacy": {
        "profileVisibility": "private",
        "showEmail": False,
        "showOnlineStatus": False
    },
    "display": {
        "timezone": "UTC",
        "dateFormat": "YYYY-MM-DD",
        "timeFormat": "24h"
    }
}


def authenticate_user(f):
    """Decorator to authenticate user via Bearer token"""
    @wraps(f)
    def decorated_function(*args, **kwargs):
        auth_header = request.headers.get('Authorization')
        
        if not auth_header:
            return jsonify({
                "error": {
                    "code": "UNAUTHORIZED",
                    "message": "Authentication required"
                }
            }), 401
        
        try:
            # Expected format: "Bearer <token>"
            token_parts = auth_header.split()
            if len(token_parts) != 2 or token_parts[0].lower() != 'bearer':
                return jsonify({
                    "error": {
                        "code": "UNAUTHORIZED",
                        "message": "Authentication required"
                    }
                }), 401
            
            token = token_parts[1]
            
            # Decode JWT token
            try:
                payload = jwt.decode(token, app.config['SECRET_KEY'], algorithms=['HS256'])
                user_id = payload.get('user_id')
                
                if not user_id:
                    return jsonify({
                        "error": {
                            "code": "UNAUTHORIZED",
                            "message": "Authentication required"
                        }
                    }), 401
                
                # Pass user_id to the route handler
                return f(user_id=user_id, *args, **kwargs)
                
            except jwt.ExpiredSignatureError:
                return jsonify({
                    "error": {
                        "code": "UNAUTHORIZED",
                        "message": "Authentication required"
                    }
                }), 401
            except jwt.InvalidTokenError:
                return jsonify({
                    "error": {
                        "code": "UNAUTHORIZED",
                        "message": "Authentication required"
                    }
                }), 401
                
        except Exception as e:
            return jsonify({
                "error": {
                    "code": "UNAUTHORIZED",
                    "message": "Authentication required"
                }
            }), 401
    
    return decorated_function


@app.route('/api/user/preferences', methods=['GET'])
@authenticate_user
def get_user_preferences(user_id):
    """
    GET /api/user/preferences endpoint
    Returns user preference settings for the authenticated user
    """
    try:
        # Retrieve user preferences from database (using mock data here)
        preferences = MOCK_USER_PREFERENCES.get(user_id)
        
        # If preferences don't exist, return default values (as per API contract notes)
        if preferences is None:
            preferences = DEFAULT_PREFERENCES.copy()
            last_updated = datetime.utcnow().isoformat() + 'Z'
        else:
            last_updated = MOCK_USER_UPDATED_TIMES.get(user_id, datetime.utcnow().isoformat() + 'Z')
        
        # Build response according to API contract
        response = {
            "userId": user_id,
            "preferences": preferences,
            "lastUpdated": last_updated
        }
        
        return jsonify(response), 200
        
    except Exception as e:
        # Log the error (use proper logging in production)
        print(f"Error retrieving preferences: {str(e)}")
        
        return jsonify({
            "error": {
                "code": "INTERNAL_ERROR",
                "message": "An unexpected error occurred"
            }
        }), 500


# Error handlers for Flask app
@app.errorhandler(404)
def not_found(error):
    return jsonify({
        "error": {
            "code": "NOT_FOUND",
            "message": "Resource not found"
        }
    }), 404


@app.errorhandler(500)
def internal_error(error):
    return jsonify({
        "error": {
            "code": "INTERNAL_ERROR",
            "message": "An unexpected error occurred"
        }
    }), 500


if __name__ == '__main__':
    app.run(debug=True, port=5000)
```
